const rows = 3;
const cols = 3;
const board = JSON.parse('{"1": [2, 2], "2": [1, 2], "3": [0, 2], "4": [2, 1], "5": [1, 1], "6": [0, 1]}');
const goal = JSON.parse('{"1": [0, 0], "2": [1, 0], "3": [2, 0], "4": [0, 1], "5": [1, 1], "6": [2, 1]}');
let selected = null;
let moves = 0;
const start = Date.now();
const boardEl = document.getElementById('board');
function render(){
  boardEl.innerHTML = '';
  for(let y=0;y<rows;y++) for(let x=0;x<cols;x++) {
    const cell = document.createElement('div');
    cell.className = 'cell ' + (board[Object.keys(board).find(k => board[k][0]==x && board[k][1]==y)] ? 'tile' : 'empty');
    const token = Object.keys(board).find(k => board[k][0]==x && board[k][1]==y);
    cell.textContent = token || '';
    cell.onclick = () => clickCell(x,y,token);
    if(selected && token===selected) cell.style.outline = '3px solid red';
    boardEl.appendChild(cell);
  }
}
function clickCell(x,y,token){
  if(token){ selected = token; render(); return; }
  if(selected){ board[selected] = [x,y]; selected = null; moves++; render(); }
}
function solved(){ return Object.keys(goal).every(k => board[k][0]===goal[k][0] && board[k][1]===goal[k][1]); }
function submitIfSolved(){
  document.getElementById('moves').value = moves;
  document.getElementById('elapsed').value = Math.floor((Date.now()-start)/1000);
  document.getElementById('board').value = JSON.stringify(board);
  if(solved()) document.getElementById('saveform').submit();
  else alert('Keep going!');
}
render();
</script>
